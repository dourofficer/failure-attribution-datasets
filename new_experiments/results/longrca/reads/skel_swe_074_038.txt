// Test for CORS allowed_origins whitespace parsing
func TestCorsAllowedOriginsWhitespace(t *testing.T) {
	tests := []struct {
		name     string
		input    string
		expected []string
	}{
		{
			name:     "comma separated",
			input:    "foo.com,bar.com",
			expected: []string{"foo.com", "bar.com"},
		},
		{
			name:     "comma with space",
			input:    "foo.com, bar.com",
			expected: []string{"foo.com", "bar.com"},
		},
		{
			name:     "space separated",
			input:    "foo.com bar.com",
			expected: []string{"foo.com", "bar.com"},
		},
		{
			name:     "tab and newline separated",
			input:    "foo.com\tbar.com\nbaz.com",
			expected: []string{"foo.com", "bar.com", "baz.com"},
		},
		{
			name:     "with leading/trailing whitespace",
			input:    "  foo.com ,  bar.com  ",
			expected: []string{"foo.com", "bar.com"},
		},
		{
			name:     "single origin",
			input:    "*",
			expected: []string{"*"},
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			// Create temp config file
			tmpFile, err := os.CreateTemp("", "flipt-test-*.yml")
			require.NoError(t, err)
			defer os.Remove(tmpFile.Name())

			configContent := fmt.Sprintf(`cors:
  enabled: true
  allowed_origins: "%s"
`, tt.input)

			_, err = tmpFile.WriteString(configContent)
			require.NoError(t, err)
			tmpFile.Close()

			// Load config
			cfg, err := Load(tmpFile.Name())
			require.NoError(t, err)

			assert.Equal(t, tt.expected, cfg.Cors.AllowedOrigins)
		})
	}
}
ENDTEST
